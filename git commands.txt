# Navigate to the project folder
cd Desktop
cd Emna
cd portfolio

# Initialize a local git repository (tracks this folder with git)
git init

# Link your local repo to a remote (GitHub) repo
# NOTE: "origin" is just a conventional name/alias — could be anything
git remote add origin https://github.com/Emna8888/portfolio.git

# Stage all files in the folder for commit
git add .

# Save a snapshot of the staged files with a message
git commit -m "add skeleton html"

# Rename the current branch to "main" (git's older local default is "master")
git branch -M main

# Push your local "main" branch to the "origin" remote for the first time
git push origin main

//Extra reference commands we discussed (didn't run yet, but good to keep):
# Rename an existing remote (e.g. if you named it wrong)
git remote rename <old-name> <new-name>

# Set git's default branch name globally for all future new repos
git config --global init.defaultBranch main

# Check which remotes are linked to your repo
git remote -v
 

//Your regular update loop going forward, once files have content:
git add .
git commit -m "describe what changed"
git push origin main
